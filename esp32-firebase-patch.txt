String b64(const uint8_t* d, size_t n) {
  size_t o = 0; mbedtls_base64_encode(NULL, 0, &o, d, n);
  uint8_t* t = (uint8_t*)ps_malloc(o + 1); if (!t) return "";
  mbedtls_base64_encode(t, o, &o, d, n); t[o] = 0; String s((char*)t); free(t); return s;
}
bool fbSend(const char* method, const String& path, const String& json) {
  WiFiClientSecure c; c.setInsecure(); HTTPClient h;
  h.begin(c, DB + path + ".json"); h.addHeader("Content-Type", "application/json");
  int r = h.sendRequest(method, json); h.end(); return r == 200;
}
